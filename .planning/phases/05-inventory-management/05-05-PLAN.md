    ```typescript
    import * as SQLite from 'expo-sqlite';

    const DB_NAME = 'breeyo-inventory-cache';

    // Initialize database and create tables on import
    export function initOfflineCache(): SQLiteDatabase {
      const db = SQLite.openDatabaseSync(DB_NAME);
      db.execSync(`
        CREATE TABLE IF NOT EXISTS barcode_cache (
          code TEXT PRIMARY KEY,
          format TEXT NOT NULL,
          item_id TEXT NOT NULL,
          item_name TEXT NOT NULL,
          item_data TEXT NOT NULL,
          synced_at INTEGER NOT NULL
        );
        CREATE TABLE IF NOT EXISTS sync_meta (
          key TEXT PRIMARY KEY,
          value TEXT NOT NULL
        );
      `);
      return db;
    }

    export class OfflineBarcodeCache {
      private db: SQLiteDatabase;

      constructor() {
        this.db = initOfflineCache();
      }

      // Sync barcode catalog from server (incremental via updatedSince)
      async syncFromServer(apiClient: ApiClient): Promise<void> {
        const lastSync = this.db.getFirstSync(
          'SELECT value FROM sync_meta WHERE key = ?', ['last_barcode_sync']
        );
        const updatedSince = lastSync ? lastSync.value : undefined;
        const response = await apiClient.get('/inventory/barcode-catalog', { params: { updatedSince } });
        for (const item of response.data.items) {
          for (const barcode of item.barcodes) {
            this.db.runSync(
              `INSERT OR REPLACE INTO barcode_cache (code, format, item_id, item_name, item_data, synced_at) VALUES (?, ?, ?, ?, ?, ?)`,
              [barcode.code, barcode.format, item.id, item.name, JSON.stringify(item), Date.now()]
            );
          }
        }
        this.db.runSync(
          `INSERT OR REPLACE INTO sync_meta (key, value) VALUES (?, ?)`,
          ['last_barcode_sync', new Date().toISOString()]
        );
      }

      // Lookup barcode offline (returns null if not found)
      lookupBarcode(code: string): CachedItem | null {
        const row = this.db.getFirstSync(
          'SELECT * FROM barcode_cache WHERE code = ?', [code]
        );
        if (!row) return null;
        return { ...row, item_data: JSON.parse(row.item_data as string) };
      }

      // Get last sync timestamp for display
      getLastSyncTime(): Date | null {
        const row = this.db.getFirstSync(
          'SELECT value FROM sync_meta WHERE key = ?', ['last_barcode_sync']
        );
        return row ? new Date(row.value as string) : null;
      }

      // Get total cached barcode count
      getCacheCount(): number {
        const row = this.db.getFirstSync('SELECT COUNT(*) as count FROM barcode_cache');
        return (row?.count as number) ?? 0;
      }
    }
    ```

    Follow RESEARCH.md Pattern 4 pending operations:

    ```typescript
    export class OfflineQueueService {
      private db: SQLiteDatabase;

      constructor() {
        this.db = initOfflineCache(); // Same database as barcode cache
        this.db.execSync(`
          CREATE TABLE IF NOT EXISTS pending_operations (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            type TEXT NOT NULL,
            payload TEXT NOT NULL,
            created_at INTEGER NOT NULL,
            synced INTEGER DEFAULT 0
          );
        `);
      }

      // Queue an operation for later sync
      enqueue(type: 'dispense' | 'adjust' | 'receive', payload: object): void {
        this.db.runSync(
          'INSERT INTO pending_operations (type, payload, created_at) VALUES (?, ?, ?)',
          [type, JSON.stringify(payload), Date.now()]
        );
      }

      // Get pending operation count
      getPendingCount(): number {
        const row = this.db.getFirstSync(
          'SELECT COUNT(*) as count FROM pending_operations WHERE synced = 0'
        );
        return (row?.count as number) ?? 0;
      }

      // Sync pending operations when online -- FIFO order, stop on first failure (Pitfall 3)
      async syncPending(apiClient: ApiClient): Promise<{ synced: number; failed: boolean }> {
        const pending = this.db.getAllSync(
          'SELECT * FROM pending_operations WHERE synced = 0 ORDER BY created_at ASC'
        );
        let synced = 0;
        for (const op of pending) {
          try {
            await apiClient.post('/inventory/sync-operation', {
              type: op.type,
              payload: JSON.parse(op.payload as string),
            });
            this.db.runSync('UPDATE pending_operations SET synced = 1 WHERE id = ?', [op.id]);
            synced++;
          } catch {
            return { synced, failed: true }; // Stop on first failure per Pitfall 3
          }
        }
        return { synced, failed: false };
      }

      // Clear synced operations (housekeeping)
      clearSynced(): void {
        this.db.runSync('DELETE FROM pending_operations WHERE synced = 1');
      }
    }
    ```
