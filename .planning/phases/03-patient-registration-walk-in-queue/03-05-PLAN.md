    **3. Create `apps/mobile/src/features/queue/hooks/useQueueSocket.ts`:**

    Socket.IO connection management with React Query cache invalidation per RESEARCH.md Pattern 2:
    ```typescript
    import { useEffect, useRef } from 'react';
    import { io, Socket } from 'socket.io-client';
    import { useQueryClient } from '@tanstack/react-query';
    import { SOCKET_EVENTS } from '@breeyo/shared/constants/socket-events';
    import * as Haptics from 'expo-haptics';
    import { useQueueUIStore } from '../store/queueUIStore';

    export function useQueueSocket(clinicId: string, token: string) {
      const socketRef = useRef<Socket | null>(null);
      const queryClient = useQueryClient();
      const soundEnabled = useQueueUIStore((s) => s.soundEnabled);
      const setOffline = useQueueUIStore((s) => s.setOffline);

      useEffect(() => {
        const API_URL = process.env.EXPO_PUBLIC_API_URL || 'http://localhost:3000';

        const socket = io(API_URL, {
          auth: { token },
          transports: ['websocket'],
          reconnection: true,
          reconnectionAttempts: Infinity,
          reconnectionDelay: 1000,
          reconnectionDelayMax: 10000,
        });

        // On new check-in from another device
        socket.on(SOCKET_EVENTS.PATIENT_CHECKED_IN, () => {
          queryClient.invalidateQueries({ queryKey: ['queue', clinicId] });
          if (soundEnabled) {
            Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium);
          }
        });

        // On status change from another device
        socket.on(SOCKET_EVENTS.QUEUE_UPDATED, () => {
          queryClient.invalidateQueries({ queryKey: ['queue', clinicId] });
          if (soundEnabled) {
            Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light);
          }
        });

        // On midnight archive
        socket.on(SOCKET_EVENTS.QUEUE_ARCHIVED, () => {
          queryClient.invalidateQueries({ queryKey: ['queue', clinicId] });
        });

        // Connection state
        socket.on('connect', () => {
          setOffline(false);
          queryClient.invalidateQueries({ queryKey: ['queue', clinicId] });
        });

        socket.on('connect_error', () => {
          setOffline(true);
        });

        socket.on('disconnect', () => {
          setOffline(true);
        });

        socketRef.current = socket;
        return () => { socket.disconnect(); };
      }, [clinicId, token]);

      return socketRef;
    }
    ```

    Status update and call-next mutations with optimistic updates:

    ```typescript
    export function useUpdateQueueStatus(clinicId: string) {
      const queryClient = useQueryClient();

      return useMutation({
        mutationFn: ({ entryId, status }: { entryId: string; status: string }) =>
          api.patch(`/api/v1/queue/${entryId}/status`, { status }),
        // Optimistic update per RESEARCH.md Pattern 3
        onMutate: async ({ entryId, status }) => {
          await queryClient.cancelQueries({ queryKey: ['queue', clinicId] });
          const previous = queryClient.getQueryData(['queue', clinicId]);
          queryClient.setQueryData(['queue', clinicId], (old: QueueBoard) => {
            // Move entry to new status group
            // ... optimistic update logic
          });
          return { previous };
        },
        onError: (_err, _vars, context) => {
          queryClient.setQueryData(['queue', clinicId], context?.previous);
          Haptics.notificationAsync(Haptics.NotificationFeedbackType.Error);
        },
        onSettled: () => {
          // Small delay to avoid flicker with Socket.IO broadcast (Pitfall 3)
          setTimeout(() => {
            queryClient.invalidateQueries({ queryKey: ['queue', clinicId] });
          }, 300);
        },
      });
    }

    export function useCallNext(clinicId: string) {
      const queryClient = useQueryClient();

      return useMutation({
        mutationFn: () => api.post('/api/v1/queue/call-next'),
        onSuccess: () => {
          queryClient.invalidateQueries({ queryKey: ['queue', clinicId] });
          Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success);
        },
      });
    }
    ```

    NOTE on optimistic updates: The `onMutate` optimistic update should reorganize the QueueBoard data locally. Move the entry from its current status group to the new status group. This gives instant UI feedback before the server responds. If the server rejects (invalid transition), `onError` rolls back.

    NOTE on Pitfall 3 (optimistic update race with Socket.IO): The `onSettled` uses a 300ms setTimeout before invalidation to avoid the flicker where Socket.IO broadcast arrives before the server commit completes.
