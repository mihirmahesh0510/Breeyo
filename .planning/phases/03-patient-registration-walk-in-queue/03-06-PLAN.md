    Stack navigator for queue-related screens:
    ```typescript
    import { createNativeStackNavigator } from '@react-navigation/native-stack';
    import { QueueScreen } from '../features/queue/screens/QueueScreen';
    import { PatientDetailScreen } from '../features/patient/screens/PatientDetailScreen';
    import { OwnerDetailScreen } from '../features/patient/screens/OwnerDetailScreen';
    import { RegisterPatientScreen } from '../features/patient/screens/RegisterPatientScreen';

    export type QueueStackParamList = {
      QueueBoard: undefined;
      PatientDetail: { petId: string };
      OwnerDetail: { ownerId: string };
      RegisterPatient: { initialMobile?: string };
    };

    const Stack = createNativeStackNavigator<QueueStackParamList>();

    export function QueueStackNavigator() {
      return (
        <Stack.Navigator screenOptions={{ headerShown: false }}>
          <Stack.Screen name="QueueBoard" component={QueueScreen} />
          <Stack.Screen name="PatientDetail" component={PatientDetailScreen} />
          <Stack.Screen name="OwnerDetail" component={OwnerDetailScreen} />
          <Stack.Screen name="RegisterPatient" component={RegisterPatientScreen} />
        </Stack.Navigator>
      );
    }
    ```

    This allows: Queue -> tap card -> PatientDetail -> OwnerDetail. Also: Queue -> CheckIn -> Register -> back to Queue.
