    ```typescript
    // Debounced search with React Query per D-25, UI-SPEC Search Behavior
    export function usePatientSearch(clinicId: string) {
      const [searchTerm, setSearchTerm] = useState('');
      const debouncedTerm = useDebounce(searchTerm, 300); // 300ms per UI-SPEC

      const searchQuery = useQuery({
        queryKey: ['patients', 'search', clinicId, debouncedTerm],
        queryFn: () => api.get(`/api/v1/patients/search?q=${encodeURIComponent(debouncedTerm)}&limit=20`),
        enabled: debouncedTerm.length >= 2, // Min 2 chars per UI-SPEC
        staleTime: 60_000,
      });

      return { searchTerm, setSearchTerm, ...searchQuery };
    }
    ```
