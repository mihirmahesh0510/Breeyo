    **2. Create `useAutoSave.ts`** -- 3-second debounced auto-save hook (per D-05, UI-SPEC Auto-Save Spec):

    ```typescript
    function useAutoSave(consultationId: string) {
      const draft = useConsultationDraftStore();
      const saveMutation = useMutation({
        mutationFn: (data: SaveDraftInput) =>
          api.patch(`/api/v1/consultations/${consultationId}/draft`, data),
        onSuccess: () => draft.markSaved(),
        onError: () => { /* retry after 5s */ },
      });

      useEffect(() => {
        if (!draft.isDirty || draft.isFinalizing) return;
        const timer = setTimeout(() => {
          draft.markSaving();
          saveMutation.mutate(serializeDraft(draft));
        }, 3000);
        return () => clearTimeout(timer);
      }, [draft.isDirty, draft.vitals, draft.subjective, draft.objective,
          draft.assessment, draft.plan, draft.careInstructions, draft.referral,
          draft.rxNotes]);

      // For finalization: force save + cancel timer
      const forceSave = async () => {
        draft.setFinalizing(true);
        if (draft.isDirty) {
          await saveMutation.mutateAsync(serializeDraft(draft));
        }
      };

      return { isSaving: saveMutation.isPending, forceSave };
    }
    ```

    Key behaviors:
    - 3-second debounce after any field change
    - If isFinalizing is true, auto-save is suppressed (End Consultation takes over)
    - forceSave() used by End Consultation flow: sets isFinalizing, saves synchronously, then navigates
    - On error: show error indicator, auto-retry after 5 seconds

    `serializeDraft(draft)` extracts only SOAP + vitals + care instructions + referral + rxNotes from the store. Does NOT include UI state (expandedSections, isDirty, etc.).

    **3. Create `useConsultationLock.ts`** -- WebSocket heartbeat for consultation lock (per D-06, UI-SPEC Lock Behavior):

    ```typescript
    function useConsultationLock(consultationId: string, vetId: string) {
      const [lockStatus, setLockStatus] = useState<{ locked: boolean; vetName?: string; stale?: boolean }>({ locked: false });

      // Check lock on mount
      useEffect(() => {
        api.get(`/api/v1/consultations/${consultationId}/lock`)
          .then(res => setLockStatus(res.data));
      }, [consultationId]);

      // Send heartbeat every 60 seconds
      useEffect(() => {
        const interval = setInterval(() => {
          api.post(`/api/v1/consultations/${consultationId}/heartbeat`)
            .catch(() => { /* log warning */ });
        }, 60000);
        return () => clearInterval(interval);
      }, [consultationId]);

      // Release lock on unmount
      useEffect(() => {
        return () => {
          // Fire-and-forget lock release on component unmount
          api.delete(`/api/v1/consultations/${consultationId}/lock`).catch(() => {});
        };
      }, [consultationId]);

      // Release on app background after 30-second delay (per UI-SPEC)
      useAppState((state) => {
        if (state === 'background') {
          const timeout = setTimeout(() => {
            api.delete(`/api/v1/consultations/${consultationId}/lock`).catch(() => {});
          }, 30000);
          return () => clearTimeout(timeout);
        }
      });

      return lockStatus;
    }
    ```

    Key behaviors:
    - Check lock status on mount
    - 60-second heartbeat interval
    - Release on unmount (navigate away or app close)
    - Release on 30-second app background (per UI-SPEC Lock Behavior)
