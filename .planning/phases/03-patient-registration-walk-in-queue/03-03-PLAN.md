    Class `QueueRepository` with constructor accepting PrismaClient:

    - `findTodayEntryForPet(clinicId: string, petId: string, today: Date)`: Query for existing QueueEntry where clinicId matches, petId matches, checkedInAt >= today, and status is WAITING or IN_CONSULT. Returns first match or null.

    - `findTodayDoneEntryForPet(clinicId: string, petId: string, today: Date)`: Same but status is DONE. For D-40 re-check-in detection.

    - `countWaiting(clinicId: string, today: Date)`: Count QueueEntry where clinicId, status WAITING, checkedInAt >= today, archivedAt is null.

    - `createEntry(data)`: `prisma.queueEntry.create({ data, include: { pet: { include: { owner: true } } } })`.

    - `findEntryById(entryId: string)`: `prisma.queueEntry.findUnique({ where: { id: entryId } })`.

    - `updateEntry(entryId: string, data)`: `prisma.queueEntry.update({ where: { id: entryId }, data, include: { pet: { include: { owner: true } } } })`.

    - `findNextWaiting(clinicId: string, today: Date)`: `prisma.queueEntry.findFirst({ where: { clinicId, status: 'WAITING', checkedInAt: { gte: today }, archivedAt: null }, orderBy: [{ isEmergency: 'desc' }, { checkedInAt: 'asc' }] })`.

    - `getQueueBoard(clinicId: string, today: Date)`: Three queries (or one query sorted/grouped):
      - inConsult: `where { clinicId, status: IN_CONSULT, archivedAt: null }`, include pet+owner, orderBy checkedInAt asc
      - waiting: `where { clinicId, status: WAITING, checkedInAt >= today, archivedAt: null }`, include pet+owner, orderBy [isEmergency desc, checkedInAt asc]
      - done: `where { clinicId, status: { in: [DONE, NO_SHOW] }, checkedInAt >= today, archivedAt: null }`, include pet+owner, orderBy completedAt desc

    - `getAverageConsultDuration(clinicId: string, days: number)`: Raw SQL query:
      ```sql
      SELECT AVG(EXTRACT(EPOCH FROM ("completedAt" - "calledAt"))) as avg_seconds
      FROM "QueueEntry"
      WHERE "clinicId" = $clinicId
        AND status = 'DONE'
        AND "calledAt" IS NOT NULL
        AND "completedAt" IS NOT NULL
        AND "completedAt" >= NOW() - INTERVAL '${days} days'
      ```
      Returns average in seconds, or null if fewer than 5 data points.

    - `archiveEntries(beforeDate: Date)`: `prisma.queueEntry.updateMany({ where: { archivedAt: null, status: { in: ['WAITING', 'DONE', 'NO_SHOW'] }, checkedInAt: { lt: beforeDate } }, data: { archivedAt: new Date() } })`. Note: D-39 says IN_CONSULT entries persist past midnight.

    **3. Create `apps/api/src/modules/queue/queue.service.ts`:**

    Class `QueueService` with constructor accepting `QueueRepository` and `Server` (Socket.IO):

    - `checkIn(params: CheckInParams)`:
      1. Parse input with `checkInSchema.parse()`
      2. Compute today = startOfDay(new Date()) in IST
      3. Check for existing active entry: `repository.findTodayEntryForPet()`. If found, throw ConflictError("Pet is already in today's queue")
      4. If reCheckIn flag not set, check for done entry: `repository.findTodayDoneEntryForPet()`. If found, throw ConflictError with code `SAME_DAY_RECHECK` so client can show confirmation dialog (D-40)
      5. If reCheckIn flag is set, skip done entry check (user confirmed)
      6. Get waiting count: `repository.countWaiting()`
      7. Create entry with position = waitingCount + 1
      8. Broadcast `SOCKET_EVENTS.PATIENT_CHECKED_IN` to `clinic:{clinicId}` room with entry data + timestamp
      9. Return entry with pet and owner included

    - `updateStatus(params: UpdateStatusParams)`:
      1. Parse with `queueStatusUpdateSchema.parse()`
      2. Find entry by ID
      3. Validate transition with `isValidTransition(entry.status, params.status)`
      4. If invalid, throw BadRequestError(`Cannot transition from ${entry.status} to ${params.status}`)
      5. Build update data: always set status + updatedAt
      6. If new status is IN_CONSULT: set treatingVetId = params.userId, calledAt = new Date()
      7. If new status is DONE or NO_SHOW: set completedAt = new Date()
      8. Update entry
      9. Broadcast `SOCKET_EVENTS.QUEUE_UPDATED` to `clinic:{clinicId}` with entry + updatedBy + timestamp
      10. Return updated entry

    - `callNext(params: CallNextParams)`:
      1. Compute today in IST
      2. Find next waiting: `repository.findNextWaiting()`. Emergency patients ordered first.
      3. If none found, throw NotFoundError("No patients waiting in queue")
      4. Call `this.updateStatus()` with IN_CONSULT status
      5. Return result

    - `getQueueBoard(params: GetQueueBoardParams)`:
      1. Compute today in IST
      2. Get board data: `repository.getQueueBoard()`
      3. Get average consult duration: `repository.getAverageConsultDuration(clinicId, 7)`
      4. Compute dynamic positions for waiting entries: enumerate waiting list, emergency patients get lower positions
      5. Compute estimated wait for each waiting entry: position x avgDuration (default 15 min = 900 seconds if avg is null or data < 5 points)
      6. Return { inConsult, waiting (with position + estimatedWait), done }

    ```typescript
    import fp from 'fastify-plugin';
    import fastifySocketIO from 'fastify-socket.io';
    import { createAdapter } from '@socket.io/redis-adapter';
    import { Redis } from 'ioredis';

    export default fp(async function socketPlugin(fastify) {
      // Redis adapter for horizontal scaling
      const pubClient = new Redis(process.env.REDIS_URL!);
      const subClient = pubClient.duplicate();

      await fastify.register(fastifySocketIO, {
        cors: {
          origin: process.env.CORS_ORIGIN || '*',
          methods: ['GET', 'POST'],
        },
        transports: ['websocket'],
        adapter: createAdapter(pubClient, subClient),
      });

      // Auth middleware: verify JWT, extract clinicId, auto-join clinic room
      fastify.io.use(async (socket, next) => {
        try {
          const token = socket.handshake.auth.token;
          if (!token) {
            return next(new Error('Authentication required'));
          }
          // Use the same JWT verification as Fastify routes
          const decoded = fastify.jwt.verify<{ userId: string; clinicId: string }>(token);
          socket.data.userId = decoded.userId;
          socket.data.clinicId = decoded.clinicId;
          // Auto-join clinic room (Pitfall 2: this also handles reconnection)
          socket.join(`clinic:${decoded.clinicId}`);
          next();
        } catch (err) {
          next(new Error('Unauthorized'));
        }
      });

      // Connection handler
      fastify.io.on('connection', (socket) => {
        // Re-join room on reconnect (Pitfall 2 from RESEARCH.md)
        const { clinicId } = socket.data;
        socket.join(`clinic:${clinicId}`);

        socket.on('disconnect', () => {
          // Room membership cleaned up automatically by Socket.IO
        });
      });
    });
    ```

    ```typescript
    import cron from 'node-cron';
    import type { PrismaClient } from '../../generated/prisma/client';
    import type { Server } from 'socket.io';
    import { SOCKET_EVENTS } from '@breeyo/shared/constants/socket-events';

    export function scheduleMidnightArchive(prisma: PrismaClient, io: Server) {
      // D-23: Auto-archive at midnight IST
      // D-39: IN_CONSULT entries persist past midnight
      cron.schedule('0 0 * * *', async () => {
        const today = new Date();
        today.setHours(0, 0, 0, 0);

        try {
          const result = await prisma.queueEntry.updateMany({
            where: {
              archivedAt: null,
              status: { in: ['WAITING', 'DONE', 'NO_SHOW'] },
              checkedInAt: { lt: today },
            },
            data: { archivedAt: new Date() },
          });

          console.log(`Midnight archive: ${result.count} entries archived`);

          // Notify all connected clients to refresh their queue
          io.emit(SOCKET_EVENTS.QUEUE_ARCHIVED, { timestamp: Date.now() });
        } catch (error) {
          console.error('Midnight archive failed:', error);
        }
      }, { timezone: 'Asia/Kolkata' }); // Pitfall 7: Always use IST timezone
    }
    ```
